#!/usr/bin/env node
import { parseArgs } from "node:util";
import { validateApprovalRecordFile } from "./lib/approval-record.mjs";

try {
  const { positionals, values } = parseArgs({ allowPositionals: true, strict: true, options: { help: {type:"boolean"}, "require-approved": { type: "boolean" }, history: { type: "boolean" } } });
  if(values.help)process.stdout.write("用法: scripts/verify-approval-record [--require-approved | --history] <record.yaml>\n默认当前校验；--history 只读历史结构，不放行。0=相应检查通过，1=失败。\n");
  else {
  if (positionals.length !== 1) throw new TypeError("用法: scripts/verify-approval-record [--require-approved | --history] <approval-record.yaml>");
  if (values.history && values["require-approved"]) throw new TypeError("--history 不能用于当前批准校验");
  validateApprovalRecordFile(positionals[0], { requireApproved: !values.history });
  process.stdout.write(!values.history ? "会签及当前用户决定校验通过\n" : "会签记录结构校验通过（不构成放行）\n");
  }
} catch (error) {
  process.stderr.write(`会签记录校验失败: ${error.message}\n`);
  process.exitCode = 1;
}
